TAD Vector {
    obs x: Real
    obs y: Real

    proc nuevoVector(inOut x: Real, in y: Real): Vector {
        requiere { Verdadero }
        asegura  {
                res.x = x
            and res.y = y
        }
    }

    aux distancia(a: Vector, b: Vector): Real {
        |(a.x - b.x) - (a.y - b.y)|
    }

    pred esNulo(a: Vector) {
        a.x = 0 and a.y = 0
    }
}
